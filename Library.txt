      'predetermine casualties or costs, or interrupt urgent action. If it does not fit this scene, disregard it. '+
      'Never print these instructions.\n'+END;
  }
  function sync(s) {
    if(!state.memory || typeof state.memory!=='object')state.memory={};
    var old=strip(state.memory.frontMemory);
    state.memory.frontMemory=old+(old&&s.pending?'\n':'')+(s.pending?cueText(s.pending):'');
  }
  function choose(s,hero,n) {
    var catalog=hero?LDD_HERO:LDD_GENERAL;
    var seq=hero?++s.heroSerial:++s.generalSerial;
    var weights=hero?(seq===1?FIRST_HERO_WEIGHT:HERO_WEIGHT):GROUP_WEIGHT;
    var k=hero?'hero':'general';
    var group=pickGroup(s,k+':group:'+seq,weights);
    var pool=catalog[group].filter(function(e){return s.recent.indexOf(e.id)===-1;});
    if(!pool.length) pool=catalog[group];
    var e=pool[rand(s,k+':item:'+seq,0,pool.length-1)];
    s.recent.push(e.id);if(s.recent.length>22)s.recent.shift();
    return {id:e.id,title:e.title,cue:e.cue,group:group,hero:hero,plannedAfter:n};
  }
  function onInput(t) {
    var s=ensure(); if(!s)return {text:t};
    if(typeof t!=='string')return {text:t};
    var c=t.trim().match(/^#(EVENTS|INCURSIONS)(?:\s+(.*))?$/i);
    if(c){
      var command=c[1].toUpperCase(),arg=(c[2]||'STATUS').trim().toUpperCase();
      state.message='';
      if(command==='INCURSIONS'){
        if(arg==='ON')s.incursions=true;
        else if(arg==='OFF'){s.incursions=false;if(s.pending&&s.pending.hero)s.pending=null;}
        else if(arg!=='STATUS')state.message='Use #INCURSIONS ON, OFF, or STATUS.';
      } else {
        if(arg==='ON'){if(!s.enabled){s.enabled=true;var a=action();if(a===null)a=s.base;s.nextGeneral=a+span(s,'resume-general:'+a,'general');s.nextHero=a+span(s,'resume-hero:'+a,s.heroSerial?'hero':'firstHero');}}
        else if(arg==='OFF') {s.enabled=false;s.pending=null;}
        else if(['QUIET','BALANCED','CHAOTIC'].indexOf(arg)!==-1){s.mode=arg.toLowerCase();
          var now=action();if(now===null)now=s.base;
          s.nextGeneral=now+span(s,'mode-general:'+now,'general');
          s.nextHero=now+span(s,'mode-hero:'+now,s.heroSerial?'hero':'firstHero');
        }
        else if(arg!=='STATUS')state.message='Use #EVENTS ON, OFF, STATUS, QUIET, BALANCED, or CHAOTIC.';
      }
      if(!state.message || !/^Use #/.test(state.message))state.message=status(s);
      s.skipAt=action();
      // A settings input is not a story turn; avoid presenting pending scene seeds here.
      s.pending=null;
      sync(s);
      // Commands cannot reliably halt the AI call via onInput; replace it with neutral OOC instruction.
      return {text:'[Out of character: Event director settings checked. Continue the current scene naturally without treating this as a story action.]'};
    }
    sync(s);
    return {text:t};
  }
  function status(s) {
    var a=action(); if(a===null)a=s.base;
    return 'Living Dungeon Director '+VERSION+' | '+(s.enabled?'ON':'OFF')+' | '+s.mode+
      ' | Incursions '+(s.incursions?'ON':'OFF')+
      ' | Next ordinary window in ~'+Math.max(0,s.nextGeneral-a)+
      ' actions; hero window in ~'+Math.max(0,s.nextHero-a)+' actions.';
  }
  function onContext(t) {var s=ensure();if(s)sync(s);return {text:t};}
  function onOutput(t) {
    var s=ensure();if(!s)return {text:t};
    var now=action(); if(now===null){s.serial++;now=s.base+s.serial;}
    if(s.lastAction===now){sync(s);return {text:t};} // replay/double hook guard
    if(integer(s.lastAction) && now<s.lastAction){
      // On undo/rewind, do not manufacture new events from stale future state.
      s.pending=null;s.lastAction=now;sync(s);return {text:t};
    }
    s.lastAction=now;
    if(s.skipAt===now){s.skipAt=null;s.pending=null;sync(s);return {text:t};}
    // A queued event was offered in this generation. Never make its outcome authoritative.
    if(s.pending)s.pending=null;
    if(!s.enabled){sync(s);return {text:t};}
    // First priority: hero director, but only at scheduled windows.
    var heroDue=s.incursions && now>=s.nextHero;
    var generalDue=now>=s.nextGeneral && now>=s.heroRestUntil;
    var globalGap=(now-s.lastEventAction)>=3;
    if(globalGap && (heroDue||generalDue)){
      var hero=!!heroDue;
      s.pending=choose(s,hero,now);
      s.lastEventAction=now;
      if(hero){
        s.nextHero=now+span(s,'next-hero:'+s.heroSerial,'hero');
        s.heroRestUntil=now+(s.pending.group==='major'?14:9);
        s.nextGeneral=Math.max(s.nextGeneral,s.heroRestUntil+2);
      } else {
        s.nextGeneral=now+span(s,'next-general:'+s.generalSerial,'general');
      }
    }
    sync(s);
    return {text:t};
  }
  return {Input:onInput,Context:onContext,Output:onOutput,Ensure:ensure,Status:status,
    _Test:{rand:rand,group:pickGroup,mode:MODES,groups:GROUP_WEIGHT,heroes:HERO_WEIGHT}};
})();
